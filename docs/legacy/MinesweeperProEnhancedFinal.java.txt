    /**
     * 创建颜色图标
     */
    private ImageIcon createColorIcon(Color color, String text) {
        int size = CELL_PIX - 6; // 稍微小于单元格
        BufferedImage image = new BufferedImage(size, size, BufferedImage.TYPE_INT_ARGB);
        Graphics2D g2d = image.createGraphics();
        
        // 设置抗锯齿
        g2d.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        
        // 绘制背景
        g2d.setColor(color);
        g2d.fillOval(2, 2, size-4, size-4);
        
        // 绘制边框
        g2d.setColor(color.darker());
        g2d.drawOval(2, 2, size-4, size-4);
        
        // 绘制文字
        if (text != null && !text.isEmpty()) {
            g2d.setColor(Color.BLACK);
            g2d.setFont(new Font("微软雅黑", Font.BOLD, 12));
            FontMetrics fm = g2d.getFontMetrics();
            int textWidth = fm.stringWidth(text);
            int textHeight = fm.getHeight();
            g2d.drawString(text, (size - textWidth) / 2, (size + textHeight) / 2 - 4);
        }
        
        g2d.dispose();
        return new ImageIcon(image);
    }

    /**
     * 检查并解锁成就
     */
    private void checkAchievements(boolean isWin) {
        // 首次胜利
        if (isWin && !achievements.get(Achievement.FIRST_WIN)) {
            unlockAchievement(Achievement.FIRST_WIN);
        }
        
        // 完美胜利（没有错误标记）
        if (isWin && wrongFlags == 0 && !achievements.get(Achievement.PERFECT_WIN)) {
            unlockAchievement(Achievement.PERFECT_WIN);
        }
        
        // 快速胜利（5回合内）
        if (isWin && currentTurn <= 5 && !achievements.get(Achievement.QUICK_WIN)) {
            unlockAchievement(Achievement.QUICK_WIN);
        }
        
        // 扫雷大师（清除100颗地雷）
        if (totalMinesCleared >= 100 && !achievements.get(Achievement.MASTER_SWEEPER)) {
            unlockAchievement(Achievement.MASTER_SWEEPER);
        }
        
        // 国王猎手（直接找到国王雷）
        // 这个成就在游戏胜利时检查具体胜利方式
        
        // 幸存者（以1点HP获胜）
        if (isWin && hp == 1 && !achievements.get(Achievement.SURVIVOR)) {
            unlockAchievement(Achievement.SURVIVOR);
        }
    }
    
    /**
     * 解锁成就
     */
    private void unlockAchievement(Achievement achievement) {
        if (!achievements.get(achievement)) {
            achievements.put(achievement, true);
            
            // 显示成就解锁通知
            SwingUtilities.invokeLater(() -> {
                JOptionPane.showMessageDialog(
                    this,
                    """
                    🎉 成就解锁！
                    
                    成就: """ + achievement.name + "\n" +
                    "描述: " + achievement.description,
                    "成就解锁",
                    JOptionPane.INFORMATION_MESSAGE
                );
            });
            
            gameHistory.add("成就解锁: " + achievement.name);
        }
    }

    /**
     * 放置两个内鬼区（满足约束：不在左上 10x10，不在国王区，互不相邻/重叠）
     */
    private void placeInformerZones() {
        int kingTL = SIZE - KING_ZONE_SIZE; // 15
        List<Point> candidates = new ArrayList<>();
        
        // 生成所有可能的左上角位置
        for (int r = 0; r <= SIZE - INFORMER_ZONE_SIZE; r++) {
            for (int c = 0; c <= SIZE - INFORMER_ZONE_SIZE; c++) {
                // 不在左上 10x10
                if (r < 10 && c < 10) continue;
                // 不与国王区重叠（即若与国王区有任何交集则排除）
                if (!(r + INFORMER_ZONE_SIZE - 1 < kingTL || c + INFORMER_ZONE_SIZE - 1 < kingTL)) continue;
                candidates.add(new Point(r, c));
            }
        }
        
        if (candidates.size() < 2) {
            gameHistory.add("警告: 内鬼区候选位置太少，使用备用位置");
            // 使用备用位置
            informerZoneTL[0] = new Point(0, 10);
            informerZoneTL[1] = new Point(10, 0);
            return;
        }
        
        // 随机取第一个
        Point a = candidates.get(rand.nextInt(candidates.size()));
        // 过滤掉与 a 相邻或重叠的候选
        List<Point> c2 = new ArrayList<>();
        for (Point p : candidates) {
            if (zonesOverlapOrAdjacent(a, p)) continue;
            c2.add(p);
        }
        
        if (c2.isEmpty()) {
            // 退化策略：顺序扫描寻找两个不重叠/不相邻的
            boolean found = false;
            outer:
            for (Point p : candidates) {
                for (Point q : candidates) {
                    if (p.equals(q)) continue;
                    if (!zonesOverlapOrAdjacent(p, q)) { 
                        informerZoneTL[0] = p; 
                        informerZoneTL[1] = q; 
                        found = true; 
                        break outer; 
                    }
                }
            }
            if (!found) {
                gameHistory.add("错误: 无法放置两个内鬼区，使用默认位置");
                informerZoneTL[0] = new Point(0, 10);
                informerZoneTL[1] = new Point(10, 0);
            }
        } else {
            informerZoneTL[0] = a;
            informerZoneTL[1] = c2.get(rand.nextInt(c2.size()));
        }
        
        gameHistory.add("内鬼区1位置: (" + informerZoneTL[0].x + "," + informerZoneTL[0].y + ")");
        gameHistory.add("内鬼区2位置: (" + informerZoneTL[1].x + "," + informerZoneTL[1].y + ")");
    }

    /**
     * 判断两个 5x5 区域是否重叠或相邻（接触边/角也视为相邻）
     */
    private boolean zonesOverlapOrAdjacent(Point a, Point b) {
        int aR0 = a.x, aC0 = a.y, aR1 = aR0 + INFORMER_ZONE_SIZE - 1, aC1 = aC0 + INFORMER_ZONE_SIZE - 1;
        int bR0 = b.x, bC0 = b.y, bR1 = bR0 + INFORMER_ZONE_SIZE - 1, bC1 = bC0 + INFORMER_ZONE_SIZE - 1;
        return !(aR1 + 1 < bR0 || bR1 + 1 < aR0 || aC1 + 1 < bC0 || bC1 + 1 < aC0);
    }

    /**
     * 在每个内鬼区内放入 10 个雷，包括 1 个内鬼核心（固定在最中间）
     */
    private void placeInformerMines() {
        for (int id = 0; id < 2; id++) {
            informerMines[id] = new HashSet<>();
            Point tl = informerZoneTL[id];
            
            // 内鬼核心固定在内鬼区最中间 (2,2) 位置
            int centerOffset = INFORMER_ZONE_SIZE / 2; // 5/2 = 2
            int rr = tl.x + centerOffset;
            int cc = tl.y + centerOffset;
            informerCore[id] = new Point(rr, cc);
            informerMines[id].add(new Point(rr, cc));
            board[rr][cc].isInformerCore = true;
            board[rr][cc].informerId = id;
            
            // 放置另外 9 颗（在该 5x5 中随机选，但不能是中心位置）
            while (informerMines[id].size() < INFORMER_ZONE_MINES + 5) {
                int r = tl.x + rand.nextInt(INFORMER_ZONE_SIZE);
                int c = tl.y + rand.nextInt(INFORMER_ZONE_SIZE);
                // 跳过中心位置（已经放了内鬼核心）
                if (r == rr && c == cc) continue;
                Point p = new Point(r, c);
                if (informerMines[id].contains(p)) continue;
                informerMines[id].add(p);
                board[r][c].isNormalMine = true; // 作为区内固定雷
                board[r][c].informerId = id;
            }
        }
        gameHistory.add("内鬼区地雷放置完成 - 每个区10颗雷，内鬼核心在中心");
    }

    /**
     * 随机把国王雷放在右下 5x5（可放在已揭开或未揭开格，之后 Phase B 会每回合重置）
     */
    private void placeKingRandom() {
        if (kingRow != -1) board[kingRow][kingCol].isKing = false;
        int start = SIZE - KING_ZONE_SIZE;
        List<Point> cand = new ArrayList<>();
        for (int r = start; r < SIZE; r++) {
            for (int c = start; c < SIZE; c++) {
                if (board[r][c].informerId == -1) cand.add(new Point(r,c));
            }
        }
        
        if (cand.isEmpty()) {
            // 极端情况，直接在 king 区任意放一个
            kingRow = start; 
            kingCol = start;
            board[kingRow][kingCol].isKing = true;
            gameHistory.add("国王雷放置警告: 使用默认位置 (" + kingRow + "," + kingCol + ")");
            return;
        }
        
        Point p = cand.get(rand.nextInt(cand.size()));
        kingRow = p.x; 
        kingCol = p.y;
        board[kingRow][kingCol].isKing = true;
        gameHistory.add("国王雷位置: (" + kingRow + "," + kingCol + ")");
    }

    /**
     * 初始在符合规则的位置放置普通雷（不包括内鬼区、左上初始区）
     */
    private void placeNormalMinesInitial() {
        // 清除非内鬼区现有的普通标记（安全）
        for (int r = 0; r < SIZE; r++) {
            for (int c = 0; c < SIZE; c++) {
                if (board[r][c].informerId == -1) board[r][c].isNormalMine = false;
            }
        }
        
        List<Point> avail = new ArrayList<>();
        for (int r = 0; r < SIZE; r++) {
            for (int c = 0; c < SIZE; c++) {
                if (r < TOPLEFT_INIT_OPEN && c < TOPLEFT_INIT_OPEN) continue; // 左上初始区不放
                if (board[r][c].informerId != -1) continue; // 内鬼区不放
                avail.add(new Point(r,c));
            }
        }
        
        Collections.shuffle(avail, rand);
        int place = Math.min(normalMineCount, avail.size());
        for (int i = 0; i < place; i++) {
            Point p = avail.get(i);
            board[p.x][p.y].isNormalMine = true;
        }
        
        // 保证国王标识正确（如果放置覆盖）
        board[kingRow][kingCol].isKing = true;
        gameHistory.add("普通雷放置完成 - 数量: " + place);
    }

    /**
     * 计算所有格的数字
     */
    private void computeAllNumbers() {
    for (int r = 0; r < SIZE; r++) {
        for (int c = 0; c < SIZE; c++) {
            board[r][c].number = computeNumberAround(r, c);
            
            // 确保内鬼区（活跃状态）没有数字0
            for (int id = 0; id < 2; id++) {
                if (informerActive[id] && 
                    r >= informerZoneTL[id].x && r < informerZoneTL[id].x + INFORMER_ZONE_SIZE &&
                    c >= informerZoneTL[id].y && c < informerZoneTL[id].y + INFORMER_ZONE_SIZE) {
                    
                    // 如果不是地雷，确保数字至少为1
                    if (!board[r][c].isKing && 
                        !(board[r][c].informerId != -1 && informerMines[board[r][c].informerId].contains(new Point(r,c))) &&
                        !board[r][c].isNormalMine) {
                        
                        if (board[r][c].number == 0) {
                            board[r][c].number = 1;
                        }
                    }
                    break;
                }
            }
        }
    }
}

    /**
     * 计算指定格子周围的地雷数量
     */
    private int computeNumberAround(int r, int c) {
        int cnt = 0;
        for (int dr = -1; dr <= 1; dr++) {
            for (int dc = -1; dc <= 1; dc++) {
                if (dr == 0 && dc == 0) continue;
                int rr = r + dr, cc = c + dc;
                if (!inBounds(rr, cc)) continue;
                if (board[rr][cc].isKing) cnt++;
                else if (board[rr][cc].isInformerCore) cnt++;
                else if (board[rr][cc].informerId != -1) {
                    // 区内固定雷：判断 informerMines 集合中是否包含
                    if (informerMines[board[rr][cc].informerId].contains(new Point(rr,cc))) cnt++;
                } else if (board[rr][cc].isNormalMine) cnt++;
            }
        }
        return cnt;
    }

    /**
     * 刷新所有按钮显示与状态
     */
    private void refreshAllButtons() {
        revealedCount = 0;
        for (int r = 0; r < SIZE; r++) {
            for (int c = 0; c < SIZE; c++) {
                JButton btn = buttons[r][c];
                Cell cell = board[r][c];
                btn.setEnabled(true);
                
                // 重置边框
                btn.setBorder(UIManager.getBorder("Button.border"));
                
                if (gameEnded) {
                    btn.setEnabled(false);
                }
                
                // 国王雷区特殊边框（蓝色边框）
                boolean isKingZone = (r >= SIZE - KING_ZONE_SIZE && c >= SIZE - KING_ZONE_SIZE);
                
                if (cell.revealed) {
                    // 已揭开的格子有黑色边框
                    btn.setBorder(BorderFactory.createLineBorder(Color.BLACK, 2));
                } else {
                    // 未揭开的格子
                    if (isKingZone) {
                        // 国王雷区：蓝色边框，始终可点击
                        btn.setBorder(BorderFactory.createLineBorder(Color.BLUE, 2));
                    } else {
                        // 普通区域：灰色边框
                        btn.setBorder(BorderFactory.createLineBorder(Color.GRAY, 1));
                    }
                }
                
                // 设置背景色
                if (cell.revealed) {
                    revealedCount++;
                    // 已点开的格子颜色加深
                    btn.setBackground(new Color(200, 200, 200));
                    
                    // 特殊区域标识
                    if (isKingZone) {
                        // 国王区
                        btn.setBackground(new Color(255, 255, 180)); // 浅黄色
                    }
                    
                    for (int id = 0; id < 2; id++) {
                        // 只有活跃的内鬼区才显示紫色
                        if (informerActive[id] && 
                            r >= informerZoneTL[id].x && r < informerZoneTL[id].x + INFORMER_ZONE_SIZE &&
                            c >= informerZoneTL[id].y && c < informerZoneTL[id].y + INFORMER_ZONE_SIZE) {
                            btn.setBackground(new Color(255, 200, 255)); // 浅紫色
                        }
                    }
                    
                    if (cell.isKing) {
                        // 国王格 - 使用图标
                        btn.setIcon(kingMineIcon);
                        btn.setText("");
                        btn.setBackground(new Color(255, 215, 0)); // 金色
                    } else {
                        // 清除图标
                        btn.setIcon(null);
                        
                        if (cell.number > 0) {
                            btn.setText(String.valueOf(cell.number));
                            // 根据数字设置颜色
                            if (cell.number <= NUMBER_COLORS.length) {
                                btn.setForeground(NUMBER_COLORS[cell.number - 1]);
                            } else {
                                btn.setForeground(Color.BLACK);
                            }
                        } else {
                            btn.setText("");
                            btn.setForeground(Color.BLACK);
                        }
                    }
                } else {
                    // 未揭开
                    btn.setBackground(UIManager.getColor("Button.background"));
                    btn.setIcon(null); // 清除图标
                    
                    if (cell.flagged) {
                        btn.setBackground(new Color(255, 200, 200)); // 浅红色
                        btn.setForeground(Color.BLACK);
                    } else {
                        btn.setText("");
                        btn.setForeground(Color.BLACK);
                    }
                }
            }
        }
        updateInfoLabels();
    }

    // ========== 玩家操作：左键 / 右键 ==========
    
    /**
     * 左键点击处理
     */
    private void onLeftClick(int r, int c) {
        totalMoves++;
        Cell cell = board[r][c];
        if (cell.revealed) return;

        // 检查点击合法性：国王雷区始终可点击，其他区域必须在已揭开格的8邻域内
        boolean isKingZone = (r >= SIZE - KING_ZONE_SIZE && c >= SIZE - KING_ZONE_SIZE);
        if (!isKingZone && !isValidClickPosition(r, c)) {
            JOptionPane.showMessageDialog(this, "只能点击未知格，且必须位于已揭开格的八邻域内（沿已揭开区域拓展）。\n国王雷区（右下5x5蓝色边框）可以随时点击。");
            return;
        }

        gameHistory.add("第" + currentTurn + "回合 - 左键点击: (" + r + "," + c + ")");

        // 如果是国王（左键揭开国王） -> 直接胜利
        if (cell.isKing) {
            // 揭开并胜利
            cell.revealed = true;
            refreshAllButtons();
            gameHistory.add("游戏胜利 - 左键揭开了国王雷！");
            // 检查国王猎手成就
            if (!achievements.get(Achievement.KING_HUNTER)) {
                unlockAchievement(Achievement.KING_HUNTER);
            }
            showGameOverDialog(true, "左键揭开了国王雷！你胜利了！");
            return;
        }

        // 如果点到内鬼区中的任意雷（包含内鬼核心或其它区内雷）
        if (cell.informerId != -1 && informerActive[cell.informerId] && informerMines[cell.informerId].contains(new Point(r,c))) {
            // 内鬼区被踩到：该内鬼区内所有雷立即爆炸，等量减少 HP，内鬼被捕获（区域失效）
            int exploded = informerMines[cell.informerId].size();
            hp -= getHpPenalty(exploded);
            minesCleared += exploded;
            totalMinesCleared += exploded;
            
            // 记录事件
            gameHistory.add("踩到内鬼区地雷 - 引爆" + exploded + "颗雷，HP -" + getHpPenalty(exploded));
            
            // 开始爆炸动画
            List<Point> explosionList = new ArrayList<>();
            for (Point p : informerMines[cell.informerId]) {
                explosionList.add(p);
            }
            startExplosionAnimation(explosionList, exploded, true);
            return;
        }

        // 如果踩到普通雷（包括国王雷区的普通雷）
        if (cell.isNormalMine) {
            gameHistory.add("踩到普通雷 - 触发连锁爆炸");
            
            // 爆炸传播：以 3x3 传播一次（不传播给国王雷）
            List<Point> explosionList = new ArrayList<>();
            explosionList.add(new Point(r, c)); // 起始点
            
            int exploded = 1; // 包括起始点
            for (int dr = -1; dr <= 1; dr++) {
                for (int dc = -1; dc <= 1; dc++) {
                    int rr = r + dr, cc = c + dc;
                    if (!inBounds(rr, cc)) continue;
                    // 国王不受传播影响
                    if (board[rr][cc].isKing) continue;
                    // 若是在内鬼区且该区有效，且该点在 informerMines 中，则该格也算爆炸
                    if (board[rr][cc].informerId != -1) {
                        int id = board[rr][cc].informerId;
                        if (informerActive[id] && informerMines[id].contains(new Point(rr,cc))) {
                            exploded++;
                            explosionList.add(new Point(rr,cc));
                        }
                    } else if (board[rr][cc].isNormalMine) {
                        exploded++;
                        explosionList.add(new Point(rr,cc));
                    }
                }
            }
            
            // 检查连锁反应成就
            if (exploded >= 5 && !achievements.get(Achievement.CHAIN_REACTION)) {
                unlockAchievement(Achievement.CHAIN_REACTION);
            }
            
            // 开始爆炸动画
            startExplosionAnimation(explosionList, exploded, false);
            return;
        }

        // 正常空格/数字格 -> 揭开（若 number==0 做标准 flood fill）
        revealCell(r,c);
        refreshAllButtons();
        
        // 检查是否胜利（通过揭开所有非雷格）
        checkWinCondition();
    }

    /**
     * 开始爆炸动画
     */
    private void startExplosionAnimation(List<Point> explosionList, int exploded, boolean isInformerZone) {
        explosionPoints = explosionList;
        explosionFrame = 0;
        
        // 禁用所有按钮
        for (int r = 0; r < SIZE; r++) {
            for (int c = 0; c < SIZE; c++) {
                buttons[r][c].setEnabled(false);
            }
        }
        
        // 创建定时器播放动画
        explosionTimer = new Timer(EXPLOSION_DELAY, (ActionEvent e) -> {
            if (explosionFrame < EXPLOSION_FRAMES) {
                // 更新当前帧
                updateExplosionFrame();
                explosionFrame++;
            } else {
                // 动画结束
                explosionTimer.stop();
                
                // 处理爆炸结果
                processExplosionResult(explosionList, exploded, isInformerZone);
            }
        });
        explosionTimer.start();
    }
    
    /**
     * 更新爆炸动画帧
     */
    private void updateExplosionFrame() {
        float progress = (float) explosionFrame / EXPLOSION_FRAMES;
        
        for (Point p : explosionPoints) {
            JButton btn = buttons[p.x][p.y];
            
            // 计算爆炸颜色（从黄色到红色）
            int red = 255;
            int green = (int) (255 * (1 - progress));
            int blue = 0;
            Color explosionColor = new Color(red, green, blue);
            
            // 创建爆炸图标
            BufferedImage img = new BufferedImage(CELL_PIX - 6, CELL_PIX - 6, BufferedImage.TYPE_INT_ARGB);
            Graphics2D g2d = img.createGraphics();
            g2d.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
            
            // 绘制爆炸效果
            g2d.setColor(explosionColor);
            int size = (int) ((CELL_PIX - 6) * (0.5 + 0.5 * progress));
            g2d.fillOval((CELL_PIX - 6 - size) / 2, (CELL_PIX - 6 - size) / 2, size, size);
            
            // 绘制爆炸中心
            g2d.setColor(Color.WHITE);
            int centerSize = size / 3;
            g2d.fillOval((CELL_PIX - 6 - centerSize) / 2, (CELL_PIX - 6 - centerSize) / 2, centerSize, centerSize);
            
            g2d.dispose();
            
            btn.setIcon(new ImageIcon(img));
            btn.setBackground(explosionColor.darker());
        }
    }
    
    /**
     * 处理爆炸结果
     */
    private void processExplosionResult(List<Point> explosionList, int exploded, boolean isInformerZone) {
        if (isInformerZone) {
            // 内鬼区爆炸
            for (Point p : explosionList) {
                int id = board[p.x][p.y].informerId;
                // 移除该区所有雷
                for (Point mp : informerMines[id]) {
                    board[mp.x][mp.y].isNormalMine = false;
                    board[mp.x][mp.y].isInformerCore = false;
                    board[mp.x][mp.y].informerId = -1;
                }
                informerMines[id].clear();
                informerActive[id] = false; // 区域失效（被捕获）
            }
            
            // 检查内鬼大师成就
            checkInformerMasterAchievement();
        } else {
            // 普通连锁爆炸
            for (Point p : explosionList) {
                board[p.x][p.y].isNormalMine = false;
                if (board[p.x][p.y].informerId != -1) {
                    // 如果来自 informerMines，需要从集合中移除
                    int id = board[p.x][p.y].informerId;
                    informerMines[id].remove(new Point(p.x,p.y));
                }
                board[p.x][p.y].informerId = -1;
                board[p.x][p.y].isInformerCore = false;
            }
        }
        
        minesCleared += exploded;
        totalMinesCleared += exploded;
        hp -= getHpPenalty(exploded);
        normalMineCount -= exploded;
        gameHistory.add("连锁爆炸 - 炸毁" + exploded + "颗地雷，HP -" + getHpPenalty(exploded));
        
        // 恢复按钮状态
        for (int r = 0; r < SIZE; r++) {
            for (int c = 0; c < SIZE; c++) {
                buttons[r][c].setEnabled(true);
                buttons[r][c].setIcon(null);
            }
        }
        
        // 进入 Phase B
        enterPhaseB();
    }

    /**
     * 右键标记处理
     */
    private void onRightClick(int r, int c) {
        totalMoves++;
        Cell cell = board[r][c];
        if (cell.revealed) return;

        // 检查点击合法性：国王雷区始终可点击，其他区域必须在已揭开格的8邻域内
        boolean isKingZone = (r >= SIZE - KING_ZONE_SIZE && c >= SIZE - KING_ZONE_SIZE);
        if (!isKingZone && !isValidClickPosition(r, c)) {
            JOptionPane.showMessageDialog(this, "只能标记未知格，且必须位于已揭开格的八邻域内（沿已揭开区域拓展）。\n国王雷区（右下5x5蓝色边框）可以随时标记。");
            return;
        }

        gameHistory.add("第" + currentTurn + "回合 - 右键点击: (" + r + "," + c + ")");

        // toggle 标记
        if (cell.flagged) {
            cell.flagged = false;
            refreshAllButtons();
            return;
        }

        // 新标记
        cell.flagged = true;

        // 判断是否命中任意雷（国王/内鬼区内雷/普通雷）
        boolean correct = false;
        if (cell.isKing) correct = true;
        else if (cell.informerId != -1 && informerActive[cell.informerId] && informerMines[cell.informerId].contains(new Point(r,c))) correct = true;
        else if (cell.isNormalMine) correct = true;

        if (correct) {
            correctFlags++;
            consecutiveCorrectFlags++;
            // 标记正确：HP +1，变为已揭开，对应地雷从盘面上移除
            hp += 1;
            cell.revealed = true;
            normalMineCount--;
            // 记录事件
            gameHistory.add("标记正确 - HP +1");
            
            // 若是国王，被右键标记直接胜利
            if (cell.isKing) {
                refreshAllButtons();
                gameHistory.add("游戏胜利 - 右键标记了国王雷！");
                // 检查国王猎手成就
                if (!achievements.get(Achievement.KING_HUNTER)) {
                    unlockAchievement(Achievement.KING_HUNTER);
                }
                showGameOverDialog(true, "你成功右键标记了国王雷！胜利！");
                return;
            }
            
            // 若是内鬼核心：立即捕获该内鬼（该内鬼区失效）
            if (cell.informerId != -1 && informerMines[cell.informerId].contains(new Point(r,c))) {
                int id = cell.informerId;
                // 移除区内雷并标记失效
                for (Point p : informerMines[id]) {
                    board[p.x][p.y].isNormalMine = false;
                    board[p.x][p.y].isInformerCore = false;
                    board[p.x][p.y].informerId = -1;
                }
                informerMines[id].clear();
                informerActive[id] = false; // 捕获
                gameHistory.add("捕获内鬼" + (id+1) + " - 获取情报");
                
                // 检查内鬼大师成就
                checkInformerMasterAchievement();
            } else {
                // 普通雷被正确标记 -> 移除（被排雷）
                cell.isNormalMine = false;
                minesCleared++;
                totalMinesCleared++;
            }
            
            computeAllNumbers();
            refreshAllButtons();
        } else {
            wrongFlags++;
            consecutiveCorrectFlags = 0;
            // 标记错误：HP -1，保持未揭开，Phase A 立即结束
            hp -= getHpPenalty(1);
            cell.flagged = false;
            gameHistory.add("标记错误 - HP -" + getHpPenalty(1) + "，Phase A 结束");
            endPhaseA(); // 触发 Phase B
        }
    }

    /**
     * 判断是否可以对 (r,c) 点击/标记：该格必须是未知格，并且位于任意已揭开格的 8 邻域内
     * 注意：国王雷区始终可点击，不受此限制
     */
    private boolean isValidClickPosition(int r, int c) {
        if (board[r][c].revealed) return false;
        
        // 国王雷区始终可点击
        if (r >= SIZE - KING_ZONE_SIZE && c >= SIZE - KING_ZONE_SIZE) {
            return true;
        }
        
        for (int dr = -1; dr <= 1; dr++) {
            for (int dc = -1; dc <= 1; dc++) {
                if (dr==0 && dc==0) continue;
                int rr = r + dr, cc = c + dc;
                if (inBounds(rr, cc) && board[rr][cc].revealed) return true;
            }
        }
        return false;
    }

    /**
     * 揭开格子（标准 flood fill：number==0 扩展）
     * 优化版：修复边缘格子洪水填充问题
     */
    private void revealCell(int r, int c) {
        if (!inBounds(r,c)) return;
        if (board[r][c].revealed) return;
        
        // 如果是地雷，不揭开（由爆炸逻辑处理）
        if (board[r][c].isKing || board[r][c].isNormalMine || 
            (board[r][c].informerId != -1 && informerActive[board[r][c].informerId] && 
             informerMines[board[r][c].informerId].contains(new Point(r,c)))) {
            return;
        }
        
        Queue<Point> queue = new LinkedList<>();
        queue.add(new Point(r, c));
        
        while (!queue.isEmpty()) {
            Point p = queue.poll();
            int pr = p.x;
            int pc = p.y;
            
            // 检查边界和是否已揭开
            if (!inBounds(pr, pc) || board[pr][pc].revealed) continue;
            
            // 揭开当前格子
            board[pr][pc].revealed = true;
            board[pr][pc].flagged = false;
            
            // 如果数字为0，继续展开周围
            if (board[pr][pc].number == 0) {
                for (int dr = -1; dr <= 1; dr++) {
                    for (int dc = -1; dc <= 1; dc++) {
                        if (dr == 0 && dc == 0) continue;
                        int nr = pr + dr;
                        int nc = pc + dc;
                        
                        // 只添加未揭开的非地雷格子
                        if (inBounds(nr, nc) && !board[nr][nc].revealed && 
                            !board[nr][nc].isKing && !board[nr][nc].isNormalMine &&
                            !(board[nr][nc].informerId != -1 && informerActive[board[nr][nc].informerId] && 
                              informerMines[board[nr][nc].informerId].contains(new Point(nr, nc)))) {
                            queue.add(new Point(nr, nc));
                        }
                    }
                }
            }
        }
    }

    /**
     * 结束 Phase A（玩家主动按按钮或被错误标记触发），进入 Phase B
     */
    private void endPhaseA() {
        phaseA = false;
        refreshAllButtons();
        enterPhaseB();
    }

    // ===== Phase B 实现（按顺序） =====
    
    /**
     * 进入 Phase B 环境阶段
     */
    private void enterPhaseB() {
        currentTurn++;
        gameHistory.add("========== 第 " + currentTurn + " 回合 Phase B 开始 ==========");
        
        // 1) 国王雷重置（随机移动到右下 5x5 区域）
        board[kingRow][kingCol].isKing = false;
        placeKingRandom();
        gameHistory.add("国王雷重置 - 新位置: (" + kingRow + "," + kingCol + ")");

        // 2) 内鬼区刷新 - 修改：整个内鬼区（5x5）的数字随机加减
        Map<Point, Integer> jitter = new HashMap<>();
        for (int id = 0; id < 2; id++) {
            if (!informerActive[id]) continue; // 已捕获的区不再影响
            
            Point tl = informerZoneTL[id];
            // 遍历整个内鬼区（5x5）
            for (int dr = 0; dr < INFORMER_ZONE_SIZE; dr++) {
                for (int dc = 0; dc < INFORMER_ZONE_SIZE; dc++) {
                    int rr = tl.x + dr, cc = tl.y + dc;
                    if (!inBounds(rr,cc)) continue;
                    // 如果该位置不是雷，则其显示的数字可能随机 +1/-1
                    boolean isMineHere = board[rr][cc].isKing
                            || (board[rr][cc].informerId != -1 && informerMines[board[rr][cc].informerId].contains(new Point(rr,cc)))
                            || board[rr][cc].isNormalMine;
                    if (!isMineHere) {
                        // 随机 +1 或 -1
                        jitter.put(new Point(rr,cc), rand.nextBoolean() ? 1 : -1);
                    }
                }
            }
        }
        
        if (!jitter.isEmpty()) {
            gameHistory.add("内鬼区数字干扰 - 影响格子数: " + jitter.size() + " (整个内鬼区)");
        }

        // 3) 视野收缩：所有已揭开且与至少一个未知格相邻的格子全部变回未揭开
        List<Point> toHide = new ArrayList<>();
        for (int r = 0; r < SIZE; r++) {
            for (int c = 0; c < SIZE; c++) {
                if (!board[r][c].revealed) continue;
                boolean hasUnknownAdj = false;
                for (int dr = -1; dr <= 1 && !hasUnknownAdj; dr++) {
                    for (int dc = -1; dc <= 1; dc++) {
                        if (dr==0 && dc==0) continue;
                        int rr = r + dr, cc = c + dc;
                        if (!inBounds(rr,cc)) continue;
                        if (!board[rr][cc].revealed) { hasUnknownAdj = true; break; }
                    }
                }
                if (hasUnknownAdj) toHide.add(new Point(r,c));
            }
        }
        
        for (Point p : toHide) {
            board[p.x][p.y].revealed = false;
            // 立即清理按钮状态
            JButton btn = buttons[p.x][p.y];
            btn.setText("");  // 清除文字
            btn.setIcon(null); // 清除图标
            btn.setBackground(UIManager.getColor("Button.background")); // 重置背景
            btn.setForeground(Color.BLACK); // 重置文字颜色
            // 注意：国王雷区有特殊边框
            boolean isKingZone = (p.x >= SIZE - KING_ZONE_SIZE && p.y >= SIZE - KING_ZONE_SIZE);
            if (isKingZone) {
                btn.setBorder(BorderFactory.createLineBorder(Color.BLUE, 2));
            } else {
                btn.setBorder(BorderFactory.createLineBorder(Color.GRAY, 1));
            }
        }
        
        gameHistory.add("视野收缩 - 隐藏格子数: " + toHide.size());

        // 4) 普通雷增殖 20%（考虑难度倍率），并在所有"未揭开格"中随机分布（不包括内鬼区，不包括左上初始区）
        int add = (int)Math.floor(normalMineCount * NORMAL_GROWTH_RATE * GROWTH_RATE_MULTIPLIER);
        normalMineCount += add;
        gameHistory.add("普通雷增殖 - 新增: " + add + "，总数: " + normalMineCount);
        
        // 清除非内鬼区的普通雷标记，重新分配
        for (int r = 0; r < SIZE; r++) {
            for (int c = 0; c < SIZE; c++) {
                if (board[r][c].informerId == -1) board[r][c].isNormalMine = false;
            }
        }
        
        List<Point> avail = new ArrayList<>();
        for (int r = 0; r < SIZE; r++) {
            for (int c = 0; c < SIZE; c++) {
                if (board[r][c].informerId != -1) continue;
                if (r < TOPLEFT_INIT_OPEN && c < TOPLEFT_INIT_OPEN) continue;
                if (board[r][c].revealed) continue; // 只在未揭开格中分配
                avail.add(new Point(r,c));
            }
        }
        
        Collections.shuffle(avail, rand);
        int place = Math.min(normalMineCount, avail.size());
        for (int i = 0; i < place; i++) {
            Point p = avail.get(i);
            board[p.x][p.y].isNormalMine = true;
        }
        
        // 保证国王标记仍然有效
        board[kingRow][kingCol].isKing = true;

        // 5) 重新计算数字并应用内鬼产生的 jitter 随机 +1/-1 效果（仅用于显示）
        computeAllNumbers();
        for (Map.Entry<Point, Integer> e : jitter.entrySet()) {
            Point p = e.getKey();
            int newNumber = board[p.x][p.y].number + e.getValue();
            // 确保干扰后内鬼区数字至少为1
            for (int id = 0; id < 2; id++) {
                if (informerActive[id] && 
                    p.x >= informerZoneTL[id].x && p.x < informerZoneTL[id].x + INFORMER_ZONE_SIZE &&
                    p.y >= informerZoneTL[id].y && p.y < informerZoneTL[id].y + INFORMER_ZONE_SIZE) {
                    
                    // 如果不是地雷，确保数字至少为1
                    if (!board[p.x][p.y].isKing && 
                        !(board[p.x][p.y].informerId != -1 && informerMines[board[p.x][p.y].informerId].contains(p)) &&
                        !board[p.x][p.y].isNormalMine) {
                        
                        newNumber = Math.max(1, newNumber);
                    }
                    break;
                }
            }
            board[p.x][p.y].number = newNumber;
        }

        // 6) 控制区检查：若已揭开格数 < 20 则失败
        int rc = 0;
        for (int r = 0; r < SIZE; r++) {
            for (int c = 0; c < SIZE; c++) {
                if (board[r][c].revealed) rc++;
            }
        }
        revealedCount = rc;
        
        if (revealedCount < 20) {
            gameHistory.add("游戏失败 - 控制区小于20: " + revealedCount);
            refreshAllButtons();
            showGameOverDialog(false, "控制区（已揭开格数）小于 20，游戏失败。");
            return;
        }

        // 保存当前棋盘状态
        saveBoardState();
        // Phase B 完成，回到 Phase A
        phaseA = true;
        refreshAllButtons();

        // 检查 HP 是否归零或以下
        if (hp <= 0) {
            gameHistory.add("游戏失败 - HP归零");
            showGameOverDialog(false, "生命值 HP 已降为 0，游戏失败。");
            return;
        }
        
        gameHistory.add("第 " + currentTurn + " 回合 Phase B 结束");
    }

    // ========== 新增功能方法 ==========

    /**
     * 根据难度获取生命值扣除倍数
     */
    private int getHpPenalty(int base) {
        return base * HP_PENALTY_MULTIPLIER;
    }

    /**
     * 保存当前棋盘状态到历史记录
     */
    private void saveBoardState() {
        int[][] state = new int[SIZE][SIZE];
        for (int r = 0; r < SIZE; r++) {
            for (int c = 0; c < SIZE; c++) {
                Cell cell = board[r][c];
                int code = 0;
                if (cell.revealed) code |= 1;
                if (cell.flagged) code |= 2;
                if (cell.isNormalMine) code |= 4;
                if (cell.isKing) code |= 8;
                if (cell.isInformerCore) code |= 16;
                code |= (cell.number << 5);
                state[r][c] = code;
            }
        }
        boardHistory.add(state);
    }

    /**
     * 检查胜利条件（揭开所有非雷格）
     */
    private void checkWinCondition() {
        boolean allNonMinesRevealed = true;
        for (int r = 0; r < SIZE; r++) {
            for (int c = 0; c < SIZE; c++) {
                Cell cell = board[r][c];
                if (!cell.isKing && !cell.isNormalMine && 
                    !(cell.informerId != -1 && informerActive[cell.informerId] && 
                      informerMines[cell.informerId].contains(new Point(r,c)))) {
                    if (!cell.revealed) {
                        allNonMinesRevealed = false;
                        break;
                    }
                }
            }
            if (!allNonMinesRevealed) break;
        }
        
        if (allNonMinesRevealed) {
            gameHistory.add("游戏胜利 - 揭开了所有非雷格！");
            showGameOverDialog(true, "你揭开了所有非雷格！胜利！");
        }
    }

    /**
     * 重新开始游戏
     */
    private void restartGame() {
        // 显示难度选择
        showDifficultySelection();
        
        // 重置游戏状态
        hp = currentDifficulty.initialHP;
        normalMineCount = currentDifficulty.initialMines;

        currentTurn = 1;
        phaseA = true;
        gameEnded = false;
        
        totalMoves = 0;
        correctFlags = 0;
        wrongFlags = 0;
        minesCleared = 0;
        totalMinesCleared = 0;
        consecutiveCorrectFlags = 0;
        startTime = System.currentTimeMillis();
        
        // 清空历史记录
        gameHistory.clear();
        boardHistory.clear();
        gameHistory.add("新游戏开始 - 时间: " + new Date());
        gameHistory.add("难度: " + currentDifficulty.name);
        
        // 根据难度设置参数
        INITIAL_NORMAL_MINES = currentDifficulty.initialMines;
        INITIAL_HP = currentDifficulty.initialHP;
        GROWTH_RATE_MULTIPLIER = currentDifficulty.growthMultiplier;
        HP_PENALTY_MULTIPLIER = currentDifficulty.hpPenaltyMultiplier;
        
        // 重新初始化游戏
        initBoard();
        placeInformerZones();
        placeInformerMines();
        placeKingRandom();
        placeNormalMinesInitial();
        computeAllNumbers();
        openInitialTopLeft();
        saveBoardState();
        
        refreshAllButtons();
        updateInfoLabels();
    }

    /**
     * 计算并显示全部地雷（游戏结束时可用）
     */
    private void revealAllMinesAndShowWrongFlags() {
        for (int r = 0; r < SIZE; r++) {
            for (int c = 0; c < SIZE; c++) {
                Cell cell = board[r][c];
                JButton btn = buttons[r][c];
                
                // 揭开所有格子
                cell.revealed = true;
                
                // 国王雷区有特殊边框
                boolean isKingZone = (r >= SIZE - KING_ZONE_SIZE && c >= SIZE - KING_ZONE_SIZE);
                
                if (isKingZone) {
                    btn.setBorder(BorderFactory.createLineBorder(Color.BLUE, 2));
                } else {
                    btn.setBorder(BorderFactory.createLineBorder(Color.BLACK, 2));
                }
                
                if (cell.isKing) {
                    btn.setIcon(kingMineIcon);
                    btn.setText("");
                    btn.setBackground(new Color(255,215,0));
                } else if (cell.isInformerCore && cell.informerId != -1 && informerActive[cell.informerId]) {
                    btn.setIcon(informerMineIcon);
                    btn.setText("");
                    btn.setBackground(new Color(255,180,180));
                } else if (cell.informerId != -1 && informerActive[cell.informerId] && 
                         informerMines[cell.informerId].contains(new Point(r,c))) {
                    btn.setIcon(mineIcon);
                    btn.setText("");
                    btn.setBackground(new Color(255, 180, 180));
                } else if (cell.isNormalMine) {
                    btn.setIcon(mineIcon);
                    btn.setText("");
                    btn.setBackground(new Color(255, 150, 150));
                } else {
                    btn.setIcon(null);
                    if (cell.number > 0) {
                        btn.setText(String.valueOf(cell.number));
                        if (cell.number <= NUMBER_COLORS.length) {
                            btn.setForeground(NUMBER_COLORS[cell.number - 1]);
                        }
                    }
                }
                
                // 标记显示
                if (cell.flagged) {
                    btn.setForeground(Color.BLACK);
                }
                
                btn.setEnabled(false);
            }
        }
    }
